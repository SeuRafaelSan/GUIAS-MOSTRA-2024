rules_version = '2';

// Regras da Central dos Guias — Mostra SESI STEAM 2026.
// • Guias entram com login anônimo do Firebase: só leem.
// • Administrador: conta e-mail/senha cujo UID existe em /admin_users.
// • Nada é apagado de verdade: exclusão de visita é lógica (campo "excluido").
service cloud.firestore {
  match /databases/{database}/documents {

    function logado() { return request.auth != null; }
    function admin() {
      return logado() && request.auth.token.firebase.sign_in_provider != 'anonymous'
        && exists(/databases/$(database)/documents/admin_users/$(request.auth.uid));
    }
    function inteiroOuNulo(v) { return v == null || (v is int && v >= 0); }

    function visitaValida(d) {
      return d.escola is string && d.escola.size() > 0 && d.escola.size() <= 120
        && d.minutos is int && d.minutos >= 1 && d.minutos <= 1440
        && d.chegadaMs is number && d.saidaPrevistaMs is number
        && d.saidaPrevistaMs == d.chegadaMs + d.minutos * 60000
        && d.periodo in ['Manhã', 'Tarde']
        && d.data is string && d.data.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && inteiroOuNulo(d.estudantes) && inteiroOuNulo(d.professores)
        && d.totalGrupos is int && d.totalGrupos >= 0
        && d.status in ['ativo', 'partiu']
        && d.excluido is bool
        && (!('observacoes' in d) || (d.observacoes is string && d.observacoes.size() <= 400));
    }

    // Base oficial de estudantes: leitura só para quem está autenticado (inclusive anônimo).
    match /students/{id} {
      allow read: if logado();
      allow write: if admin();
    }
    match /student_presentations/{id} {
      allow read: if logado();
      allow write: if admin();
    }

    match /school_visits/{id} {
      allow read: if logado();
      allow create, update: if admin() && visitaValida(request.resource.data);
      allow delete: if false;
    }
    match /visit_groups/{id} {
      allow read: if logado();
      allow write: if admin();
    }
    match /event_maps/{id} {
      allow read: if logado();
      allow write: if admin() && request.resource.size() < 1048000;
    }
    match /notifications/{id} {
      allow read: if logado();
      allow write: if admin();
    }
    match /system_settings/{id} {
      allow read: if logado();
      allow write: if admin();
    }
    match /audit_logs/{id} {
      allow read, create: if admin();
      allow update, delete: if false;
    }
    match /admin_users/{uid} {
      allow read: if logado() && request.auth.uid == uid;
      allow write: if false; // cadastrado só pelo console do Firebase
    }
  }
}
